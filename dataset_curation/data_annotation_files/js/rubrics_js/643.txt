When rewriting an oracle that constructs dynamic XPath from strings, copy this exact escaping helper syntax:
```javascript
function escapeXPathString(str) {
  const s = String(str);
  if (s.indexOf("'") === -1) return "'" + s + "'";
  if (s.indexOf('"') === -1) return '"' + s + '"';
  const parts = s.split("'").map(p => "'" + p + "'");
  return 'concat(' + parts.join(', "\'", ') + ')';
}
```
